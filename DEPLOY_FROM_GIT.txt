# Команды для деплоя с GitHub на сервер
# База данных сохранится автоматически (она в volume)

# 1. Подключитесь к серверу:
ssh root@72.60.182.59

# 2. Перейдите в директорию проекта:
cd /root/crm-

# 3. Если git еще не инициализирован, инициализируйте:
git init
git remote add origin https://github.com/olbat1os/crm-.git

# 4. Получите изменения и переключитесь на ветку:
git fetch origin
git checkout update-december-2024 || git checkout -b update-december-2024 origin/update-december-2024
git pull origin update-december-2024

# 5. Проверьте что база данных на месте (она должна сохраниться):
docker exec crm--db-1 psql -U crm_user -d crm_db -c "SELECT COUNT(*) FROM business;"

# 6. Пересоберите и перезапустите контейнеры:
docker-compose down
docker-compose up -d --build

# 7. Проверьте статус:
docker-compose ps
docker-compose logs --tail=30 crm-app
