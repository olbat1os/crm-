Команды для загрузки исправления обратной связи на сервер:

# 1. Обновите ветку на сервере:
cd ~/crm-
git fetch origin
git checkout fix/array-validation-december-2024
git pull origin fix/array-validation-december-2024

# 2. Скопируйте обновленный файл в контейнер:
CONTAINER="crm--crm-app-1"
docker cp /root/crm-/app/main.py ${CONTAINER}:/app/app/main.py

# 3. Удалите кэш Python:
docker exec ${CONTAINER} find /app -name "*.pyc" -delete
docker exec ${CONTAINER} find /app -name "__pycache__" -type d -exec rm -rf {} + 2>/dev/null || true

# 4. Перезапустите контейнер:
docker-compose restart crm-app

# 5. Проверьте логи:
docker-compose logs --tail=30 crm-app

# 6. Проверьте, что endpoint работает:
docker exec ${CONTAINER} curl -s -X PUT http://localhost:8000/api/bookings/1 -H "Content-Type: application/json" -d '{"feedback":"test"}' | head -n 5

Или проще пересоздать контейнер:
docker-compose up -d --force-recreate --no-deps crm-app
