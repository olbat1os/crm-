# Команды для загрузки SSL скриптов на сервер
# Выполните эти команды в PowerShell на вашем компьютере

# ============================================
# ВАРИАНТ 1: PowerShell (Windows)
# ============================================

# 1. Загрузите скрипты на сервер
scp renew_ssl.sh root@72.60.182.59:/root/crm-/
scp setup_ssl_auto_renew.sh root@72.60.182.59:/root/crm-/

# 2. Установите права на выполнение (подключитесь к серверу)
ssh root@72.60.182.59 "cd crm- && chmod +x renew_ssl.sh setup_ssl_auto_renew.sh"

# ============================================
# ВАРИАНТ 2: Использование готового скрипта
# ============================================

# В PowerShell выполните:
.\UPLOAD_SSL_SCRIPTS.ps1

# ============================================
# ВАРИАНТ 3: Ручная загрузка через WinSCP или FileZilla
# ============================================

# Сервер: 72.60.182.59
# Пользователь: root
# Пароль: (ваш пароль)
# Путь на сервере: /root/crm-/
# Загрузите файлы:
#   - renew_ssl.sh
#   - setup_ssl_auto_renew.sh

# Затем подключитесь по SSH и выполните:
#   cd crm-
#   chmod +x renew_ssl.sh setup_ssl_auto_renew.sh

# ============================================
# После загрузки на сервере выполните:
# ============================================

# 1. Обновите SSL сертификат (замените yourdomain.com и your@email.com):
cd crm-
./renew_ssl.sh yourdomain.com your@email.com

# 2. (Опционально) Настройте автоматическое обновление:
./setup_ssl_auto_renew.sh yourdomain.com your@email.com
